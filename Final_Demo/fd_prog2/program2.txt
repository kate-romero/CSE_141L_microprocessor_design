// get input
LOAD R5
MOVT R1
LOAD R4
MOVT R2

// get registers ready
// sign mask R5 = b10000000
MOVI 4
MOVT R5
MOVI 8
LSHL R5
MOVT R5
// exponent mask R6 = b01111100
MOVI 3
MOVT R7
MOVI 4
MOVT R6
MOVI 15
LSHL R7
BORR R6
MOVT R6
// exponent compare R7 = 00010001 (17)
MOVI 14
ADDR R7
MOVT R7
// exponent shift R13 = 2
MOVI 2
MOVT R13
// exponent too large comparison R14 = 19
ADDR R7
MOVT R14
// shift for exp too large R15 = 1
MOVI 1
MOVT R15

// grab sign bit
MOVF R1
BAND R5
MOVT R3

// clear sign bit to prevent false leading 1
BNOT R5
MOVF R1
BAND R5
MOVT R1
BNOT R5

// grab exponent + bias
BAND R6
LSHR R13
MOVT R8

// check exponent too large
CMPR R14
BRLT $exponent_suff_small
// 1-fill upper byte
MOVF R5
BNOT R0
BORR R3
STOR R7
// 1-fill lower byte
LSHL R15
ADDR R15
STOR R6
HALT

// mask off exponent + bias
$exponent_suff_small:
BNOT R6 //$exponent_suff_small
MOVF R1
BAND R6
BNOT R6
MOVT R1

// unhide leading 1
MOVI 4
BORR R1
MOVT R1

// (exponent + bias) > 17 ? << : >>
MOVF R8
CMPR R7
BRLT $exp_gt_17

// mantissa << (exp + bias) - 17
// exp = exp - 17
SUBR R7
MOVT R11
// build lower byte
MOVF R2
LSHL R11
STOR R6
// build upper byte
MOVF R1
LSHL R11
MOVT R1
MOVI 8
SUBR R11
MOVT R11
MOVF R2
LSHR R11
BORR R1
BORR R3
STOR R7
HALT

$exp_gt_17:
// mantissa >> 17 - (exp + bias)
MOVF R7 //$exp_gt_17
SUBR R8
MOVT R11
// build upper byte
MOVF R1
LSHR R11
BORR R3
STOR R7
// built lower byte
MOVF R2
LSHR R11
MOVT R2
MOVI 8
SUBR R11
MOVT R11
MOVF R1
LSHL R11
BORR R2
STOR R6
HALT
