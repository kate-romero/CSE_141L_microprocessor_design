// get upper and lower bytes
LOAD M1
MOVT R1
LOAD M0
MOVT R2

// prep registers
// sign mask R5 = b10000000
MOVI b0100  //4
MOVT R5
MOVI b1000  //8
LSHL R5
MOVT R5
// max exponent R6 = 6
MOVI b0110  //6
MOVT R6
// exponent bias R7 = 15
MOVI b1111  //15
MOVT R7
// zero reg for comparing R15 = 0
MOVI 0
MOVT R15
// one reg for decrementing and shifting R14 = 1
MOVI 1
MOVT R14

// grab sign bit
MOVF R1
BAND R5
MOVT R3

// clear sign bit to prevent false leading 1
BNOT R5
MOVF R1
BAND R5
MOVT R1
BNOT R5

// determine which byte contains leading 1
MOVF R1
CMPR R15
BNEQ $label_0   // branch if leading 1 in upper byte
MOVF R2
CMPR R15
BNEQ $label_2    // branch if leading 1 in lower byte
DONE

// put lower byte in R1
MOVT R1     // $label_2

// find leading 1
MOVF R1     // re-get leading 1 byte    $label_0
BAND R5
CMPR R15    // #49.if ((reg_imp & reg_sign_mask) != 0)
BNEQ $label_1

// decrement exponent
MOVF R6
SUBR R14
MOVT R6

// sign mask logic shift right
MOVF R5
LSHR R14
MOVT R5

// loop back
CMPR R15    // R0 ? 0    #47.while (reg_sign_mask)
BNEQ $label_0

// reset R1
LOAD M1
// clear sign bit
BNOT R5
MOVF R1
BAND R5
MOVT R1
BNOT R5
